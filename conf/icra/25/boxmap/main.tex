\section{Overview: Core Questions and Answers}

\paragraph{(1) What is the problem?} 
\begin{itemize}
  \item Navigation in Complex and \textbf{Unknown} (a.k.a Unmapped or Partial Sensed) Environments
  \item Robot Navigation in \textbf{Resource-Constrained} Scenarios
  \item \textbf{Topological Graph Generation} and \textbf{Real-Time Updating} -> Topological Graph Prediction (what actually do)
\end{itemize}

\paragraph{(2) Why need to solve this problem?}
\begin{itemize}
  \item Traditional standard navigation methods require maintenance of \textit{detailed} environment representations.
  \item Maintaining detailed environment representations is \textit{resource-expensive}.
  \item Demands for running within resource-constrained scenarios.
\end{itemize}

\paragraph{(3) How is it different from prev.?}
\begin{itemize}
  \item Compared with traditional \textit{pixel-wise semantic segmentation}, \textit{wall entities identification \& clustering}:
  \begin{itemize}
    \item Utilizing \textbf{deep learning} to learn from \textbf{prior experience}. 
    \begin{itemize}
      \item ?: \textit{Learning fr. prior} can be achieved solely through DL?
    \end{itemize}
  \end{itemize}
  \item Compared with \textit{Anchors based Bounding Boxes Prediction + CNN-RNN (or CNN-GCN)} based \textit{Corners Detection}:
  \begin{itemize}
    \item \textbf{Anchor-Free}:\textbf{Transformer encoder-decoder architecture} to \textbf{eliminate anchor generation};
    \item \textbf{Box-based (set-based) loss} to \textbf{eliminate Non-Maximum Suppression (NMS)};
    \item In Summarization, an \textbf{end-to-end} DEtection TRansformer (DETR) based method to reduce reliance on \textbf{hand-crafted} components (NMS, anchor generation, etc).
  \end{itemize}
  \item Compared with other \textit{DETR + regression/classification} in Topological Generation:
  \begin{itemize}
    \item Using \textbf{bounding boxes} directly (no need for \textit{vertices}) as \textbf{primitives} of environment (map) (\textit{core});
    \item Thus, no \textit{post-processing components} for vertices obtaining.
  \end{itemize}
\end{itemize}

\paragraph{(4) Why is it better than prev.? (Advantages)} 
\begin{itemize}
  \item \textbf{BoxMap} (bounding boxes as primitives, within a DETR-like framework) enables:
  \begin{itemize}
    \item Utilizing bounding boxes directly (combined with \textbf{L2 loss} (which is designed for \textit{TSDF-based} representations)) \textbf{eliminates vertices-based manual labeling};
    \item \textbf{No need} for \textit{multi-resolution representations}, as well as \textit{extra post-processing} like regression/classification.
  \end{itemize}
  \item BoxMap + \textbf{TSDF-based Representation} enables \textbf{Resource-Efficient};
  \item Utilizing \textbf{Hierarchical loss} enables Better \textbf{Small Details Detection};
  \item Enabling \textit{downstream decision making tasks} (planning \& navigation, e.g. A* search based methods) to generate \textbf{shorter trajectories}.
\end{itemize}

\paragraph{(5) What is the approach itself?}
\begin{itemize}
  \item A \textbf{DEtection TRansformer (DETR)} based \textbf{end-to-end} architecture to \textit{incrementally} generate \textbf{box-based} (using \textbf{TSDF-based} representation) topological map.
  \begin{itemize}
    \item Novelty:
    \begin{itemize}
      \item BoxMap + TSDF -> Resource-Efficient
      \item \textit{Fine-tuning} with (\textit{Prior Predicted}, Current Measurements) -> Learning fr. \textit{Prior} (\textit{Recursive})
      \item Hierarchical Loss -> Highlight small details.
    \end{itemize}
    \item BoxMap Components: $cf.$ \cref{tab:boxmap_components}
  \end{itemize}
\end{itemize}

\section{Open Questions}

\begin{enumerate}
    \item \textbf{Validity of Historical Predictions:} 
    How are historical predictions (Prior Predicted) used for fine-tuning obtained? And what characteristics make a historical prediction indeed beneficial for incremental mapping stability?
    \item \textbf{Generalizability:}
    While utlizing TSDF-based representations achieve resource efficiency in structured environments, how can this framework generalize to large-scale or non-Manhattan outdoor scenes without increasing memory requirements or losing geometric detail?
\end{enumerate}
