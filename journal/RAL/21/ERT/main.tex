\section{Overview: Core Questions and Answers}
                           
\paragraph{(1) What is the problem?} 
\begin{itemize}
  \item \textbf{Accurate Robot Dynamics Modeling:} Learning robot dynamics \textit{from data} for systems whose states evolve on a \textbf{matrix Lie group} (e.g., the $SE(3)$ manifold for rigid bodies), while ensuring the model respects \textbf{physical} laws (e.g. \textbf{energy conservation} and \textbf{kinematic constraints}).
  \item \textbf{Stabilization and Generalizability:} Developing a \textbf{unified control policy} capable of stabilization and \textbf{trajectory tracking} for different rigid-body mobile \textbf{UGVs}, \textbf{UAVs} or \textbf{UUVs} under \textbf{same} control design.
\end{itemize}

\paragraph{(2) Why need to solve this problem?}
\begin{itemize}
  \item Accurate dynamics models are critical for \textbf{safe}, \textbf{stable} control and \textbf{generalization (generalizing?)} to novel operational conditions.
  \item Necessity deriving from current situation (methods):
  \begin{itemize}
    \item Traditional models (e.g. hand-designed) derived from first principles often \textit{over-simplify} the system, resulting in modeling errors that simple parameter tuning cannot correct.
    \item In contrast, standard \textit{black-box} machine learning models require \textit{large amounts of training data \& computational} time (\textit{impractical in mobile} robotics applications) and often fail to infer physical laws (like kinematic constraints and energy conservation).
  \end{itemize}
\end{itemize}

\paragraph{(3) How is it different from prev.?}
\begin{itemize}
  \item Compared to structured learning:
  \begin{itemize}
    \item Previous Hamiltonian/Lagrangian neural networks primarily operated on vector-valued states in Euclidean space and relied on parameterizations like Euler angles, which suffer from singularities.
    \item This work embeds matrix Lie groups in a \textbf{higher-dimensional} space, allowing to train the model via the widely used \textbf{neural ODE network on Euclidean space}.
  \end{itemize}
  \item Compared to Lie group networks:
  \begin{itemize}
    \item Existing Lie group neural ODEs (for learning dynamics and normalizing flows) mostly focused on preserving structure during backpropagation, but they did provide physical system descriptions.
    \item This work embeds \textbf{physical system descriptions} via incorporating \textbf{Hamiltonian mechanics}.
  \end{itemize}
  \item In summarization, this work introduces a \textbf{port-Hamiltonian formulation} capable of embedding any general \textbf{matrix Lie group constraint} and explicitly models \textbf{energy-dissipation elements} (such as \textbf{friction} or \textbf{drag forces}).
\end{itemize}

\paragraph{(4) Why is it better than prev.? (Advantages)} 
\begin{itemize}
  \item \textbf{Physics and Geometry Preservation:} By embedding port-Hamiltonian mechanics and matrix Lie group constraints directly into the neural ODE, the model automatically obeys \textbf{energy conservation} and \textbf{geometric constraints} ($SO(3)$ constraints and $SE(3)$ constraints) during \textit{long-term trajectory predictions}.
  \item \textbf{Efficiency:} The structured model \textbf{converges faster} and achieves significantly \textbf{lower training loss and prediction errors} compared to \textit{unstructured} or \textit{black-box networks}.
  \item \textbf{Unified Model-based Controller:} It directly designs a unified, \textbf{system-agnostic} and \textbf{model-based} controller using \textit{interconnection and damping assignment passivity-based control (IDA-PBC)}, allowing \textbf{stable trajectory tracking} for fully-actuated and underactuated robots \textbf{without requiring a priori knowledge} of system parameters.
\end{itemize}

\paragraph{(5) What is the approach itself?}
\begin{itemize}
  \item Port-Hamiltonian neural ODE network on a Lie group
  \begin{itemize}
    \item 4 separate neural networks to \textbf{approximate physical components}:
    \begin{itemize}
      \item kinetic energy $T$
      \item potential energy $V$
      \item energy dissipation matrix $D$
      \item input matrix $B$
    \end{itemize}
    \item The networks dictate the continuous-time dynamics within an ODE solver to minimize the errors between predicted and true states.
  \end{itemize}
  \item Unified Control Policy: It applies an energy-shaping and damping-injection strategy to reshape the learned system's energy landscape, so its global minimum aligns with the desired tracking trajectory.
\end{itemize}

\paragraph{(6) What are the applications of it?} 
\begin{itemize}
  \item The approach (designed \textit{dynamics models}) is applicable to rigid-body \textbf{mobile} robots, such as \textbf{unmanned ground vehicles (UGVs)}, \textbf{unmanned aerial vehicles (UAVs)} and \textbf{unmanned underwater vehicles (UUVs)}.
  \item Experiments both in \textit{simulation} and \textit{\textbf{real} PX4 quadrotors}.
\end{itemize}
